const express = require('express'); const multer = require('multer'); const { exec } = require('child_process'); const fs = require('fs-extra'); const path = require('path'); const cors = require('cors'); const app = express(); app.use(cors()); app.use(express.static('public')); const upload = multer({ dest: 'uploads/' }); const jobs = {}; app.post('/build-apk', upload.fields([{ name: 'htmlFile' }, { name: 'icon' }]), async (req, res) => { try { const rawName = req.body.appName || 'MyApp'; const cleanName = rawName.replace(/[^a-zA-Z0-9]/g, ''); const jobId = Date.now().toString(); const projectDir = path.join(__dirname, 'builds', jobId); if (!req.files.htmlFile) { return res.status(400).json({ error: 'HTML ফাইল আবশ্যক।' }); } jobs[jobId] = { status: 'processing', message: 'প্রজেক্ট সেটআপ করা হচ্ছে...' }; res.json({ jobId }); // ব্যাকগ্রাউন্ড বিল্ড প্রসেস (async () => { try { // ১. Cordova প্রজেক্ট তৈরি await runCommand(`npx cordova create "${projectDir}" com.build.${cleanName.toLowerCase()} "${cleanName}"`); // ২. HTML ফাইল কপি করা const wwwDir = path.join(projectDir, 'www'); await fs.emptyDir(wwwDir); await fs.copy(req.files.htmlFile[0].path, path.join(wwwDir, 'index.html')); // ৩. আইকন যোগ করা (যদি দেওয়া থাকে) if (req.files.icon) { const iconPath = path.join(projectDir, 'res', 'icon.png'); await fs.ensureDir(path.join(projectDir, 'res')); await fs.copy(req.files.icon[0].path, iconPath); } // ৪. অ্যান্ড্রয়েড প্ল্যাটফর্ম যুক্ত করে APK কম্পাইল করা jobs[jobId].message = 'অ্যান্ড্রয়েড APK কম্পাইল করা হচ্ছে...'; await runCommand(`cd "${projectDir}" && npx cordova platform add android && npx cordova build android`); const apkPath = path.join(projectDir, 'platforms', 'android', 'app', 'build', 'outputs', 'apk', 'debug', 'app-debug.apk'); if (await fs.pathExists(apkPath)) { jobs[jobId] = { status: 'done', apkPath, cleanName }; } else { jobs[jobId] = { status: 'error', message: 'APK আউটপুট তৈরি হয়নি।' }; } } catch (err) { console.error(err); jobs[jobId] = { status: 'error', message: 'কম্পাইলেশন ব্যর্থ হয়েছে।' }; } finally { // আপলোডকৃত টেম্পোরারি ফাইল ক্লিনআপ if (req.files.htmlFile) fs.remove(req.files.htmlFile[0].path); if (req.files.icon) fs.remove(req.files.icon[0].path); } })(); } catch (e) { res.status(500).json({ error: e.message }); } }); // বিল্ডের বর্তমান অবস্থা চেক করার API app.get('/check-status/:id', (req, res) => { const job = jobs[req.params.id]; if (!job) return res.status(404).json({ status: 'error', message: 'ফাইল খুঁজে পাওয়া যায়নি।' }); res.json(job); }); // তৈরি হওয়া APK ডাউনলোড করার API app.get('/download-apk/:id', async (req, res) => { const job = jobs[req.params.id]; if (job && job.status === 'done' && await fs.pathExists(job.apkPath)) { res.download(job.apkPath, `${job.cleanName}.apk`, async () => { // ডাউনলোড শেষে ফোল্ডার রিমুভ await fs.remove(path.join(__dirname, 'builds', req.params.id)); delete jobs[req.params.id]; }); } else { res.status(404).send('ফাইলটি পাওয়া যায়নি।'); } }); function runCommand(cmd) { return new Promise((resolve, reject) => { exec(cmd, (error, stdout, stderr) => { if (error) reject(stderr || stdout); else resolve(stdout); }); }); } const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log(`Build Server running on port ${PORT}`));
